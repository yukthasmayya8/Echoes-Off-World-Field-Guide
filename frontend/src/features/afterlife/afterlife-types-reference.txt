export type LocationConfidence = 'exact' | 'approximate' | 'unknown';

export interface Source {
  id: string;
  organization: string;
  title: string;
}

export interface TimelineEvent {
  id: string;
  title: string;
  date: string | null;                 // "YYYY", "YYYY-MM" or "YYYY-MM-DD"; null if not documented
  dateEnd?: string | null;             // NEW: for date ranges
  datePrecision?: 'day' | 'month' | 'year' | 'range' | 'unknown';
  description: string;
  category:
    | 'launch' | 'arrival' | 'landing' | 'deployment' | 'first_activity'
    | 'milestone' | 'discovery' | 'final_communication'
    | 'end_of_operations' | 'final_disposition';
  sourceIds: string[];
  chapterId?: string;                  // NEW (optional): links an event to a Storybook chapter
}

export interface Chapter {
  id: string;
  title: string;
  body?: string[];                     // paragraphs; chapter index 4 = "What Remains"
  sourceIds?: string[];
}

export interface Artifact {
  id: string;
  name: string;
  body: 'Moon' | 'Mars';
  mission?: { name: string; relationship: string };
  status: {
    operational: string;
    communication?: string;            // NEW: spec 12.4 needs a third status
    physicalDisposition: string;       // "unknown" is a valid value
  };
  location: {
    description: string;               // "unknown" is a valid value
    confidence: LocationConfidence;
    uncertaintyExplanation?: string;   // NEW
    coordinates?: { lat: number; lon: number };  // only if verified
    sourceIds?: string[];
  };
  chapters: Chapter[];
  timeline: TimelineEvent[];
  discoveries: { id: string; title: string }[];
  sources: Source[];
}
